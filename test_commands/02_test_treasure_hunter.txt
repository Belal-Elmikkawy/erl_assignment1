#####################################################################
 PART A - node alone, fake /odometry/filtered and /detected_marker
 Gazebo must NOT be running (it would also publish these topics).
#####################################################################

IMPORTANT: the yaml sets use_sim_time: true. Without Gazebo there is no
/clock, the clock stays at 0 and the control timer never fires. So in
Part A always add  -p use_sim_time:=false

---------------------------------------------------------------------
 A1  Run the node
---------------------------------------------------------------------
ros2 run erl1_assignment1 treasure_hunter_node.py --ros-args --params-file $PARAMS \
  -p use_sim_time:=false
#   expect: Treasure hunter ready. Instructions (rad): {1: -1.57, 2: 3.14, 3: 1.57}, final marker 4

---------------------------------------------------------------------
 A2  Watch the output (keep these open the whole time)
---------------------------------------------------------------------
ros2 topic echo /cmd_vel
ros2 topic echo /marker_reached
ros2 topic echo /treasure_found

---------------------------------------------------------------------
 A3  WAIT_ODOM: no odometry -> robot still
---------------------------------------------------------------------
#   /cmd_vel shows linear.x 0.0, angular.z 0.0

---------------------------------------------------------------------
 A4  Fake odometry, yaw = 0  -> FORWARD
---------------------------------------------------------------------
ros2 topic pub -r 20 /odometry/filtered nav_msgs/msg/Odometry \
  "{pose: {pose: {orientation: {z: 0.0, w: 1.0}}}}"
#   log:      State WAIT_ODOM -> FORWARD
#   /cmd_vel: linear.x 0.25, angular.z 0.0

---------------------------------------------------------------------
 A5  Marker too small (far) -> ignored, stays in FORWARD
---------------------------------------------------------------------
ros2 topic pub -r 20 /detected_marker std_msgs/msg/Float32MultiArray \
  "{data: [320.0, 381.36, 1.0, 320.0, 240.0, 20.0]}"
#   s = 20 < min_marker_side_px (40) -> no state change. Ctrl+C this pub.

---------------------------------------------------------------------
 A6  Marker 1 on the right, far -> SERVO (align first, then approach)
---------------------------------------------------------------------
ros2 topic pub -r 20 /detected_marker std_msgs/msg/Float32MultiArray \
  "{data: [320.0, 381.36, 1.0, 400.0, 240.0, 100.0]}"
#   log: Marker 1 found, start visual servoing / State FORWARD -> SERVO
#   expected numbers:
#     err_u = 400-320 = +80  ->  w = -0.003*80 = -0.24   (turn right, negative)
#     err_s = 166-100 = +66  ->  v = 0.004*66 = 0.264 * (1 - 80/100) = ~0.05
#   log: SERVO id 1: u-cx=+80px  sd-s=+66px  -> v=+0.05 w=-0.24
#   Ctrl+C this pub.

# marker centred, still far -> full approach speed, no rotation
ros2 topic pub -r 20 /detected_marker std_msgs/msg/Float32MultiArray \
  "{data: [320.0, 381.36, 1.0, 320.0, 240.0, 100.0]}"
#   v = 0.264 (clamped to max_linear 0.3 not needed), w = 0.0

# too close (s > sd) -> must go BACKWARDS
ros2 topic pub -r 20 /detected_marker std_msgs/msg/Float32MultiArray \
  "{data: [320.0, 381.36, 1.0, 320.0, 240.0, 200.0]}"
#   v = 0.004*(166-200) = -0.136
#   Ctrl+C this pub.

---------------------------------------------------------------------
 A7  Lost marker -> stop after 0.6 s, back to FORWARD after 2 s
---------------------------------------------------------------------
# (with A6 stopped and nothing published on /detected_marker)
#   log: Marker 1 lost, searching again / State SERVO -> FORWARD
#   Now repeat the centred pub of A6 to get back into SERVO.

---------------------------------------------------------------------
 A8  Marker 1 reached -> REACHED -> ROTATE -90 deg
---------------------------------------------------------------------
ros2 topic pub -r 20 /detected_marker std_msgs/msg/Float32MultiArray \
  "{data: [320.0, 381.36, 1.0, 320.0, 240.0, 166.0]}"
#   log: Marker 1 REACHED
#        State SERVO -> REACHED
#   /marker_reached: data: 1
#   after 1 s:  Instruction of marker 1: rotate -90 deg / State REACHED -> ROTATE
#   /cmd_vel: linear.x 0.0, angular.z -0.6  (fake yaw is stuck at 0, so it keeps turning)
#   KEEP this pub running: marker 1 is now "done" and must be ignored.

---------------------------------------------------------------------
 A9  Pretend the rotation is done: yaw = -90 deg -> FORWARD
---------------------------------------------------------------------
# stop the A4 odometry pub, then:
ros2 topic pub -r 20 /odometry/filtered nav_msgs/msg/Odometry \
  "{pose: {pose: {orientation: {z: -0.7071068, w: 0.7071068}}}}"
#   log: State ROTATE -> FORWARD
#   /cmd_vel: linear.x 0.25, angular.z ~0
#   Marker 1 is still published (A8) but there is NO new SERVO -> "used once" works.
#   Ctrl+C the A8 pub.

# heading hold check: yaw drifts to -80 deg, the target is -90 -> must turn right
ros2 topic pub -r 20 /odometry/filtered nav_msgs/msg/Odometry \
  "{pose: {pose: {orientation: {z: -0.6427876, w: 0.7660444}}}}"
#   /cmd_vel angular.z = 1.0 * (-10 deg) = ~ -0.17
#   then go back to the -90 deg pub above.

---------------------------------------------------------------------
 A10 Marker 2 -> rotate 180 deg (target yaw = +90 deg)
---------------------------------------------------------------------
ros2 topic pub -r 20 /detected_marker std_msgs/msg/Float32MultiArray \
  "{data: [320.0, 381.36, 2.0, 320.0, 240.0, 166.0]}"
#   /marker_reached: 2, log: rotate +180 deg, ROTATE with angular.z +0.6
#   Ctrl+C, then end the rotation with yaw = +90 deg:
ros2 topic pub -r 20 /odometry/filtered nav_msgs/msg/Odometry \
  "{pose: {pose: {orientation: {z: 0.7071068, w: 0.7071068}}}}"
#   log: State ROTATE -> FORWARD

---------------------------------------------------------------------
 A11 Marker 3 -> rotate +90 deg (target yaw = 180 deg)
---------------------------------------------------------------------
ros2 topic pub -r 20 /detected_marker std_msgs/msg/Float32MultiArray \
  "{data: [320.0, 381.36, 3.0, 320.0, 240.0, 166.0]}"
#   /marker_reached: 3, rotate +90 deg, angular.z +0.6. Ctrl+C, then yaw = 180 deg:
ros2 topic pub -r 20 /odometry/filtered nav_msgs/msg/Odometry \
  "{pose: {pose: {orientation: {z: 1.0, w: 0.0}}}}"
#   log: State ROTATE -> FORWARD   (tests the +-pi wrap)

---------------------------------------------------------------------
 A12 Marker 4 -> FOUND, robot stops for ever
---------------------------------------------------------------------
ros2 topic pub -r 20 /detected_marker std_msgs/msg/Float32MultiArray \
  "{data: [320.0, 381.36, 4.0, 320.0, 240.0, 166.0]}"
#   /marker_reached: 4
#   /treasure_found: data: 'Treasure found! (marker 4)'
#   log: State REACHED -> FOUND,  /cmd_vel all zeros from now on.

---------------------------------------------------------------------
 A13 Robustness
---------------------------------------------------------------------
# malformed message -> WARN, ignored
ros2 topic pub --once /detected_marker std_msgs/msg/Float32MultiArray "{data: [1.0, 2.0, 3.0]}"
#   WARN: Malformed /detected_marker message ignored

# two markers in the same frame -> must lock on the BIGGEST (closest) unused one
# (restart the node with A1 + A4 first)
ros2 topic pub -r 20 /detected_marker std_msgs/msg/Float32MultiArray \
  "{data: [320.0, 381.36, 2.0, 100.0, 240.0, 60.0, 3.0, 500.0, 240.0, 90.0]}"
#   log: Marker 3 found, start visual servoing

# Ctrl+C on the node -> last /cmd_vel must be zero


#####################################################################
 PART B - full hunt in Gazebo
#####################################################################

---------------------------------------------------------------------
 B1  Simulation
---------------------------------------------------------------------
ros2 launch erl1_assignment1 spawn_assignment.launch.py

---------------------------------------------------------------------
 B2  Check the inputs of the hunter
---------------------------------------------------------------------
ros2 topic hz /odometry/filtered                       # ~30 Hz (EKF)
ros2 topic echo --once /odometry/filtered --field pose.pose.orientation
#   at start: z ~ 0, w ~ 1 (yaw 0, facing +x)

---------------------------------------------------------------------
 B3  Both nodes together
---------------------------------------------------------------------
ros2 launch erl1_assignment1 hunt.launch.py
#   explicit params file:
#   ros2 launch erl1_assignment1 hunt.launch.py params_file:=$PARAMS

---------------------------------------------------------------------
 B4  Watch
---------------------------------------------------------------------
ros2 topic echo /marker_reached
ros2 topic echo /treasure_found
ros2 run rqt_image_view rqt_image_view /marker_image   # one new image per reached marker
ros2 run rqt_image_view rqt_image_view /aruco_debug    # live view

# yaw in degrees, live (check the 90/180 deg rotations end close to the target)
python3 -c "
import math,rclpy
from nav_msgs.msg import Odometry
rclpy.init(); n=rclpy.create_node('yaw_print')
def cb(m):
    q=m.pose.pose.orientation
    print(f'yaw = {math.degrees(math.atan2(2*(q.w*q.z+q.x*q.y),1-2*(q.y**2+q.z**2))):+7.2f} deg')
n.create_subscription(Odometry,'/odometry/filtered',cb,10); rclpy.spin(n)"

# expected sequence with this world (boxes: 1 at (4,0), 2 at (3,-4), 3 at (3,5), 4 at (-1,4)):
#   FORWARD -> SERVO 1 -> REACHED 1 -> ROTATE -90 (yaw -90)
#   FORWARD -> SERVO 2 -> REACHED 2 -> ROTATE 180 (yaw +90)
#   FORWARD (marker 1 may be visible again - must be ignored)
#           -> SERVO 3 -> REACHED 3 -> ROTATE +90 (yaw 180)
#   FORWARD -> SERVO 4 -> REACHED 4 -> FOUND

---------------------------------------------------------------------
 B5  Restart a run without restarting Gazebo
---------------------------------------------------------------------
# Ctrl+C hunt.launch.py, then reset the world (robot back to the origin):
gz service -s /world/aruco_world/control --reqtype gz.msgs.WorldControl \
  --reptype gz.msgs.Boolean --timeout 3000 --req 'reset: {all: true}'
# The EKF keeps its old pose after a reset: restart the whole simulation (B1)
# if the yaw printed above is not ~0.

---------------------------------------------------------------------
 B6  Record for the video / debugging
---------------------------------------------------------------------
ros2 bag record /detected_marker /cmd_vel /odometry/filtered /marker_reached /treasure_found /marker_image
